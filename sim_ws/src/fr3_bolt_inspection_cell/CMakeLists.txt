cmake_minimum_required(VERSION 3.16)
project(fr3_bolt_inspection_cell)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gazebo_ros2_control REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
add_library(fr3_inspection_contact_system SHARED src/contact_system.cpp)
target_compile_features(fr3_inspection_contact_system PUBLIC cxx_std_17)
target_include_directories(fr3_inspection_contact_system PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(fr3_inspection_contact_system PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(fr3_inspection_contact_system ${GAZEBO_LIBRARIES})
ament_target_dependencies(fr3_inspection_contact_system gazebo_ros2_control hardware_interface pluginlib rclcpp)
pluginlib_export_plugin_description_file(gazebo_ros2_control contact_system_plugins.xml)
add_library(fr3_inspection_grasp SHARED src/sim_grasp.cpp)
target_compile_features(fr3_inspection_grasp PUBLIC cxx_std_17)
target_include_directories(fr3_inspection_grasp PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(fr3_inspection_grasp PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(fr3_inspection_grasp ${GAZEBO_LIBRARIES})
ament_target_dependencies(fr3_inspection_grasp gazebo_ros rclcpp std_srvs sensor_msgs)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS fr3_inspection_grasp fr3_inspection_contact_system LIBRARY DESTINATION lib)
install(PROGRAMS scripts/inspection_task scripts/inspection_panel DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_linked_finger_servo test/test_linked_finger_servo.cpp)
  target_compile_features(test_linked_finger_servo PUBLIC cxx_std_17)
  add_test(NAME linked_finger_servo COMMAND test_linked_finger_servo)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(inspection_algorithms test/test_core.py)
  ament_add_pytest_test(inspection_cartesian test/test_cartesian.py)
  ament_add_pytest_test(inspection_planning_io test/test_ros_io_planning.py)
  ament_add_pytest_test(inspection_descent_kinematics test/test_descent_kinematics.py)
  ament_add_pytest_test(inspection_retry_feedback test/test_retry_feedback.py)
endif()
ament_package()
